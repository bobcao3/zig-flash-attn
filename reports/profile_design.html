Each warp logs zone begins and ends as 4-byte records.

Mark and capture

const Zone = enum(u7) { tile, scores, qk_mma };
const Trace = profile.Recorder(Zone, .{ .sink = .lanes });
fn scoreTile(params: Params, args: Args) void {
    var trace: Trace = .init(params.profile);
    defer trace.finish();
    trace.begin(.scores);
    defer trace.end(.scores);
    const slot = acquireKeys(args);
    trace.begin(.qk_mma);
    defer trace.end(.qk_mma);
    multiplyScores(args, slot);
}
# zig build -Dprofile
from zigfa import profile
with profile.capture() as cap:
    flash_attn_func(q, k, v)
cap.save("fwd.jsonl.gz")
print(cap.summary(by="sm"))

All 32 lanes of a warp call begin and end together; without -Dprofile, both compile to nothing.
